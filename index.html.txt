  <script>
    let rawData = [];

    // Tải danh sách file từ data.json
    async function initApp() {
      try {
        const res = await fetch('data.json');
        rawData = await res.json();
        renderDocs(rawData);
      } catch (err) {
        document.getElementById('appContent').innerHTML = '<p style="color:red">Không thể tải data.json. Hãy kiểm tra đường dẫn file.</p>';
      }
    }

    // Hiển thị danh sách Card
    function renderDocs(data) {
      const container = document.getElementById('appContent');
      container.innerHTML = '';

      data.forEach(group => {
        if (group.items.length === 0) return;
        
        const catElem = document.createElement('div');
        catElem.className = 'category-group';
        
        let itemsHTML = group.items.map(item => `
          <div class="card" onclick="openDoc('${item.file}', '${item.title}')">
            <div class="card-code">${item.code}</div>
            <div class="card-title">${item.title}</div>
            <div class="card-meta">${item.issuer} • ${item.articles}</div>
          </div>
        `).join('');

        catElem.innerHTML = `
          <div class="category-header">${group.category}</div>
          ${itemsHTML}
        `;
        container.appendChild(catElem);
      });
    }

    // Đọc & Đưa nội dung file .docx vào Modal
    async function openDoc(filePath, title) {
      document.getElementById('docModalTitle').innerText = title;
      document.getElementById('docViewer').innerHTML = '<p><i class="fa-solid fa-spinner fa-spin"></i> Đang đọc file Word (.docx)...</p>';
      document.getElementById('docModal').style.display = 'block';

      try {
        const response = await fetch(filePath);
        const arrayBuffer = await response.arrayBuffer();
        const result = await mammoth.convertToHtml({ arrayBuffer: arrayBuffer });
        document.getElementById('docViewer').innerHTML = result.value || '<p>File rỗng hoặc không đọc được.</p>';
      } catch (err) {
        document.getElementById('docViewer').innerHTML = `<p style="color:#ff6b6b">Lỗi: Không tải được file <b>${filePath}</b>. Hãy đảm bảo bạn đã copy file .docx vào đúng thư mục docs/.</p>`;
      }
    }

    function closeModal() {
      document.getElementById('docModal').style.display = 'none';
    }

    // Thanh tìm kiếm nhanh
    function filterDocs() {
      const query = document.getElementById('searchInput').value.toLowerCase();
      const filtered = rawData.map(group => ({
        category: group.category,
        items: group.items.filter(item => 
          item.code.toLowerCase().includes(query) ||
          item.title.toLowerCase().includes(query) ||
          item.issuer.toLowerCase().includes(query)
        )
      }));
      renderDocs(filtered);
    }

    window.onload = initApp;
